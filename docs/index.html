CLI & Rust SDK Quickstart

CLI Usage
# Benchmark inference throughput
mlx-vlm bench --arch paligemma --num-tokens 64

# Describe an image with PaliGemma
mlx-vlm generate \
  --image ./document.png \
  --prompt "Transcribe the table data into JSON" \
  --temperature 0.2

# Inspect model architecture parameters
mlx-vlm info --arch qwen2vl
Rust SDK Integration
use mlx_vlm_rs::{VlmPipeline, VlmConfig, VlmArchitecture, GenerationConfig};

fn main() -> anyhow::Result<()> {
    let config = VlmConfig::preset(VlmArchitecture::PaliGemma);
    let mut pipeline = VlmPipeline::new(config)?;

    let img = pipeline.load_image("chart.png")?;
    let prompt_tokens = vec![257152, 101, 102]; // <image> + prompt

    pipeline.generate(Some(&img), &prompt_tokens, &GenerationConfig::default(), |token| {
        print!("{}", token);
        true
    })?;
    Ok(())
}
